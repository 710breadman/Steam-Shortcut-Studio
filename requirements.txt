# Pillow enables JPEG/WebP artwork previews and the modern shell's artwork
# thumbnails. The classic window still runs without it.

# The modern shell is the default interface, so this is a production
# dependency, not an optional one. main.py falls back to the classic window if
# it is somehow missing from a build, but a correct build always includes it.
customtkinter==6.0.0
# Bundled Linux builds use certifi so HTTPS artwork/API calls do not depend on
# a distro-specific CA bundle being visible from the PyInstaller runtime.
certifi>=2024.8.30
